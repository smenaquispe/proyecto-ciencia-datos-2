statsbomb D WITH stats AS (
              SELECT
                quantile_cont(duration, 0.25) AS q1,
                quantile_cont(duration, 0.75) AS q3
              FROM events_fact
              WHERE duration IS NOT NULL
            )
            SELECT
              COUNT(*) AS outliers_duration
            FROM events_fact, stats
            WHERE duration IS NOT NULL
              AND (duration < q1 - 1.5 * (q3 - q1) OR duration > q3 + 1.5 * (q3 - q1));
